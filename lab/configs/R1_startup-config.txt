! =====================================================================
!  Router R1 - Multi-VLAN gateway (router-on-a-stick)
!  Project: Enterprise ARP Analysis & Threat Detection
!  Team   : Stuti Zingade, Ritika Tulsyan, Akkshar Ranjan
!
!  VLAN 10 users   192.168.10.0/24  gw .1
!  VLAN 20 servers 192.168.20.0/24  gw .1
!  VLAN 30 guest   192.168.30.0/24  gw .1
!  G0/0 is an 802.1Q trunk to SW1; one subinterface per VLAN.
! =====================================================================

enable
configure terminal
!
hostname R1
no ip domain-lookup
ip domain-name enterprise.lab
service password-encryption
enable secret Cisco@Enable123
username netadmin privilege 15 secret Cisco@Admin123
!
! ---------------------------------------------------------------------
! Trunk uplink to SW1 with one subinterface (gateway) per VLAN
! ---------------------------------------------------------------------
interface GigabitEthernet0/0
 description TRUNK to SW1 (802.1Q, all VLANs)
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 description VLAN10 USERS gateway
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
!
interface GigabitEthernet0/0.20
 description VLAN20 SERVERS gateway
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
!
interface GigabitEthernet0/0.30
 description VLAN30 GUEST gateway
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
!
! Age dynamic ARP faster than the 4 h default so stale entries clear.
 arp timeout 1200
!
! ---------------------------------------------------------------------
! Anti-poisoning anchors: static (permanent) ARP for the server so a
! spoofed unicast reply cannot overwrite it. Fill in the real MAC.
! (Example MAC from the lab dataset shown; replace with the captured one.)
! ---------------------------------------------------------------------
! arp 192.168.20.30 001a.2b00.a030 ARPA   ! Enterprise Server (VLAN 20)
!
! Optional: restrict guest VLAN from reaching the server VLAN
ip access-list extended GUEST-ISOLATION
 deny ip 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
!
interface GigabitEthernet0/0.30
 ip access-group GUEST-ISOLATION in
!
! ---------------------------------------------------------------------
! Secure management
! ---------------------------------------------------------------------
crypto key generate rsa modulus 2048
ip ssh version 2
line console 0
 logging synchronous
 login local
line vty 0 4
 transport input ssh
 login local
!
service timestamps log datetime msec
logging buffered 16384 informational
!
end
copy running-config startup-config
!

!    show ip interface brief
!    show ip arp
!    show arp
!    debug arp                 ! watch live ARP request/reply
! =====================================================================
