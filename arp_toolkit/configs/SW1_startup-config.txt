! =====================================================================
!  Switch SW1 - Startup Configuration (Catalyst 2960-Series)
!  Project: Enterprise ARP Analysis
!  Role   : Access switch for VLAN 10 (192.168.10.0/24)
!  Team   : Stuti Zingade, Ritika Tulsyan, Akkshar Ranjan
!
!  Port map:
!    Fa0/1  -> R1  (uplink / default gateway)
!    Fa0/2  -> PC1 (192.168.10.10)
!    Fa0/3  -> PC2 (192.168.10.20)
!    Fa0/4  -> Enterprise Server (192.168.10.30)
!    Fa0/5  -> Network Printer   (192.168.10.40)
! =====================================================================

enable
configure terminal
!
hostname SW1
no ip domain-lookup
service password-encryption
enable secret Cisco@Enable123
username netadmin privilege 15 secret Cisco@Admin123
!
! ---------------------------------------------------------------------
! VLAN definition and management interface
! ---------------------------------------------------------------------
vlan 10
 name USERS
!
interface Vlan10
 description Management SVI
 ip address 192.168.10.2 255.255.255.0
 no shutdown
!
ip default-gateway 192.168.10.1
!
! ---------------------------------------------------------------------
! Uplink to R1 - trunk-less single VLAN, treat as access in VLAN 10.
! No port-security here (gateway must be reachable by all hosts).
! ---------------------------------------------------------------------
interface FastEthernet0/1
 description UPLINK to R1 (Default Gateway)
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast disable
!
! ---------------------------------------------------------------------
! Access ports to end devices.
! Port security mitigates MAC flooding (CAM-table overflow) and limits
! each port to its single legitimate host, which also makes a rogue
! device harder to introduce for ARP spoofing.
!   - max 1 (or 2 for printer+mgmt) MAC addresses
!   - sticky learning writes the learned MAC into the config
!   - violation restrict: drop offending frames + log, keep port up
! ---------------------------------------------------------------------
interface range FastEthernet0/2 - 5
 description ACCESS - VLAN10 end device
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 switchport port-security aging time 60
 switchport port-security aging type inactivity
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! ---------------------------------------------------------------------
! DAI (Dynamic ARP Inspection) - the strongest switch-side defense
! against ARP poisoning. Requires DHCP snooping (or static ARP ACLs).
! Available on 2960 running LAN Base. Enable if your platform supports it.
! ---------------------------------------------------------------------
ip dhcp snooping
ip dhcp snooping vlan 10
no ip dhcp snooping information option
!
ip arp inspection vlan 10
ip arp inspection validate src-mac dst-mac ip
!
! Trust the uplink to R1 (and any DHCP server port); untrust access ports
interface FastEthernet0/1
 ip dhcp snooping trust
 ip arp inspection trust
!
! Rate-limit ARP on access ports to blunt flooding attacks
interface range FastEthernet0/2 - 5
 ip arp inspection limit rate 15
!
! ---------------------------------------------------------------------
! Shut unused ports (Fa0/6 - Fa0/24, Gi0/1 - Gi0/2)
! ---------------------------------------------------------------------
interface range FastEthernet0/6 - 24
 description UNUSED
 shutdown
!
interface range GigabitEthernet0/1 - 2
 description UNUSED
 shutdown
!
! ---------------------------------------------------------------------
! Secure management access
! ---------------------------------------------------------------------
crypto key generate rsa modulus 2048
ip ssh version 2
!
line console 0
 logging synchronous
 exec-timeout 10 0
 login local
!
line vty 0 4
 transport input ssh
 login local
!
service timestamps log datetime msec
logging buffered 16384 informational
!
end
copy running-config startup-config
!

!    show ip arp inspection statistics
