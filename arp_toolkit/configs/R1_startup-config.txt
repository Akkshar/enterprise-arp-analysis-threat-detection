! =====================================================================
!  Router R1 - Startup Configuration
!  Project: Enterprise ARP Analysis
!  Role   : Default gateway for VLAN 10 (192.168.10.0/24)
!  Team   : Stuti Zingade, Ritika Tulsyan, Akkshar Ranjan
!  Target : Cisco ISR (IOS 15.x). Tested layout for Packet Tracer / GNS3.
! =====================================================================

enable
configure terminal
!
! ---------------------------------------------------------------------
! Basic device identity & hardening
! ---------------------------------------------------------------------
hostname R1
!
no ip domain-lookup
ip domain-name enterprise.lab
!
! Encrypt plaintext passwords in the config
service password-encryption
!
! Privileged EXEC secret (replace with your own strong secret)
enable secret Cisco@Enable123
!
! Local admin account for SSH
username netadmin privilege 15 secret Cisco@Admin123
!
! ---------------------------------------------------------------------
! LAN interface - default gateway for the single VLAN
! ---------------------------------------------------------------------
interface GigabitEthernet0/0
 description LAN - VLAN10 Default Gateway (192.168.10.0/24)
 ip address 192.168.10.1 255.255.255.0
 no shutdown
!
! Keep the ARP cache fresh but not too aggressive. Default IOS ARP
! timeout is 4 hours (14400 s); shorten it so stale entries age out
! faster in a lab where devices move.
 arp timeout 1200
!
! ---------------------------------------------------------------------
! OPTIONAL static ARP entries for critical infrastructure.
! A static (permanent) ARP entry cannot be overwritten by a spoofed
! unicast reply, so it is a simple anti-poisoning measure for the
! server and printer. Fill in the real MAC addresses from your lab.
! ---------------------------------------------------------------------
! arp 192.168.10.30 0000.0000.0030 ARPA   ! Enterprise Server
! arp 192.168.10.40 0000.0000.0040 ARPA   ! Network Printer
!
! ---------------------------------------------------------------------
! Secure remote management (SSH v2 only, no Telnet)
! ---------------------------------------------------------------------
crypto key generate rsa modulus 2048
ip ssh version 2
!
line console 0
 logging synchronous
 exec-timeout 10 0
 login local
!
line vty 0 4
 transport input ssh
 login local
 exec-timeout 10 0
!
! ---------------------------------------------------------------------
! Logging & time (useful when correlating ARP events with captures)
! ---------------------------------------------------------------------
service timestamps log datetime msec
service timestamps debug datetime msec
logging buffered 16384 informational
!
! ---------------------------------------------------------------------
! Save and exit
! ---------------------------------------------------------------------
end
copy running-config startup-config
!
! =====================================================================
!  Verification commands (run from privileged EXEC):
!    show ip interface brief
!    show arp
!    show ip arp 192.168.10.10
!    debug arp            ! watch live ARP request/reply on R1
!    clear arp-cache      ! force re-resolution (for troubleshooting)
! =====================================================================
